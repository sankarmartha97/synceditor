📋 Quick SQL Queries to Try:
═══════════════════════════════════════════════════════════

-- View all users
SELECT * FROM users;

-- View all canvases with owner names
SELECT c.*, u.name as owner_name 
FROM canvases c 
JOIN users u ON c.owner_id = u.id;

-- Count widgets per canvas
SELECT c.name, COUNT(w.id) as widget_count
FROM canvases c
LEFT JOIN widgets w ON c.id = w.canvas_id
GROUP BY c.id, c.name;

-- View recent widget changes
SELECT * FROM widget_versions 
ORDER BY created_at DESC 
LIMIT 10;

-- View canvas collaborators
SELECT c.name as canvas_name, 
       u.name as collaborator_name,
       cc.role
FROM canvas_collaborators cc
JOIN canvases c ON cc.canvas_id = c.id
JOIN users u ON cc.user_id = u.id;
